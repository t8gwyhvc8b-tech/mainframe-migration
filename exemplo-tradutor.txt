"""Configure the local mainframe MCP server for Devin on Windows."""

from __future__ import annotations

import json
import os
from pathlib import Path
import platform
import re
import shutil
import subprocess
import sys
from typing import Any


DEFAULT_VERSION = "0.1.13"
EXTENSION_DIRECTORY_PREFIX = (
    "mainframe-migration-toolkit.mainframe-migration-toolkit-"
)
REQUIRED_TOOLS = {
    "mainframe_preflight",
    "mainframe_dependency_graph",
    "mainframe_get_callers",
    "mainframe_resolve_copybook",
    "mainframe_impact_analysis",
    "mainframe_run_migration_tool",
    "mainframe_query_artifact",
}


class SetupError(RuntimeError):
    """Raised when the offline Devin setup cannot be completed."""


def running_on_windows() -> bool:
    """Return whether the setup is running on Windows."""
    return platform.system() == "Windows"


def supported_python() -> bool:
    """Return whether the interpreter meets the toolkit requirement."""
    return sys.version_info >= (3, 10)


def is_toolkit_runtime(path: Path, expected_version: str) -> bool:
    """Return whether a directory contains the expected bundled runtime."""
    package_main = path / "mainframe_modernization_toolkit" / "__main__.py"
    if not package_main.is_file():
        return False

    manifest_path = path / "manifest.json"
    if not manifest_path.is_file():
        return True
    try:
        manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        return False
    return (
        isinstance(manifest, dict)
        and manifest.get("packageVersion") == expected_version
    )


def find_toolkit_runtime(root: Path, expected_version: str) -> Path:
    """Locate toolkit sources in the repository or an installed VSIX."""
    explicit = os.environ.get("MAINFRAME_TOOLKIT_RUNTIME")
    candidates = [
        root / "src",
        root / "extension" / "python-runtime",
    ]
    if explicit:
        candidates.insert(0, Path(explicit).expanduser())

    extension_roots: list[Path] = []
    configured_extensions = os.environ.get("VSCODE_EXTENSIONS")
    if configured_extensions:
        extension_roots.append(Path(configured_extensions).expanduser())
    extension_roots.append(Path.home() / ".vscode" / "extensions")

    extension_name = EXTENSION_DIRECTORY_PREFIX + expected_version
    for extension_root in extension_roots:
        candidates.append(extension_root / extension_name / "python-runtime")
        if extension_root.is_dir():
            candidates.extend(
                extension / "python-runtime"
                for extension in sorted(
                    extension_root.glob(EXTENSION_DIRECTORY_PREFIX + "*"),
                    reverse=True,
                )
            )

    checked: set[Path] = set()
    for candidate in candidates:
        resolved = candidate.resolve()
        if resolved in checked:
            continue
        checked.add(resolved)
        if is_toolkit_runtime(resolved, expected_version):
            return resolved

    locations = "\n  - ".join(str(path) for path in checked)
    raise SetupError(
        "runtime local do toolkit não encontrado. Locais verificados:\n  - "
        + locations
        + "\nDefina MAINFRAME_TOOLKIT_RUNTIME se o VSIX estiver em outro local."
    )


def adapt_skill(source: Path, target: Path) -> None:
    """Copy and adapt the Claude skill for Devin."""
    if not (source / "SKILL.md").is_file():
        raise SetupError(f"skill MCP não encontrada em {source / 'SKILL.md'}")

    target.mkdir(parents=True, exist_ok=True)
    shutil.copytree(source, target, dirs_exist_ok=True)
    skill_path = target / "SKILL.md"
    content = skill_path.read_text(encoding="utf-8")

    missing = sorted(tool for tool in REQUIRED_TOOLS if tool not in content)
    if missing:
        raise SetupError(
            "a skill MCP de origem não contém as tools esperadas: "
            + ", ".join(missing)
        )

    content, substitutions = re.subn(
        r"(?m)^name:\s*mainframe-jcl-migration\s*$",
        "name: mainframe-jcl-migration-devin",
        content,
        count=1,
    )
    if substitutions != 1:
        raise SetupError("não foi possível adaptar o nome da skill")

    if not re.search(r"(?m)^triggers:", content):
        content, substitutions = re.subn(
            r"(?m)^(argument-hint:[^\n]*\n)",
            r"\1triggers:\n  - user\n  - model\n",
            content,
            count=1,
        )
        if substitutions != 1:
            raise SetupError("não foi possível adicionar os triggers")

    skill_path.write_text(content, encoding="utf-8")


def load_config(path: Path) -> dict[str, Any]:
    """Load an existing MCP configuration without discarding other servers."""
    if not path.exists():
        return {}
    try:
        config = json.loads(path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as error:
        raise SetupError(f"JSON inválido em {path}: {error}") from error
    if not isinstance(config, dict):
        raise SetupError(f"{path} precisa conter um objeto JSON")
    return config


def write_mcp_config(path: Path, toolkit_source: Path) -> None:
    """Register the local source tree using the current Python interpreter."""
    config = load_config(path)
    servers = config.setdefault("mcpServers", {})
    if not isinstance(servers, dict):
        raise SetupError(f"'mcpServers' em {path} precisa ser um objeto")

    servers["mainframe-toolkit"] = {
        "command": sys.executable,
        "args": [
            "-m",
            "mainframe_modernization_toolkit",
            "mcp",
            "serve",
        ],
        "env": {
            "PYTHONPATH": str(toolkit_source),
        },
    }

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(
        json.dumps(config, indent=2, ensure_ascii=True) + "\n",
        encoding="utf-8",
    )
    temporary.replace(path)
    json.loads(path.read_text(encoding="utf-8"))


def toolkit_environment(toolkit_source: Path) -> dict[str, str]:
    """Build an environment that imports the toolkit from the repository."""
    environment = os.environ.copy()
    current = environment.get("PYTHONPATH")
    environment["PYTHONPATH"] = (
        str(toolkit_source)
        if not current
        else str(toolkit_source) + os.pathsep + current
    )
    return environment


def validate_runtime(toolkit_source: Path, expected_version: str) -> None:
    """Verify the local MCP dependency and toolkit version without downloading."""
    environment = toolkit_environment(toolkit_source)
    dependency_check = subprocess.run(
        [
            sys.executable,
            "-c",
            "from mainframe_modernization_toolkit.mcp_server import serve_mcp",
        ],
        env=environment,
        capture_output=True,
        text=True,
        check=False,
    )
    if dependency_check.returncode != 0:
        raise SetupError(
            "o Python disponível não contém a dependência mcp compatível; "
            "sem download, ela precisa estar previamente instalada"
        )

    version_check = subprocess.run(
        [
            sys.executable,
            "-m",
            "mainframe_modernization_toolkit",
            "--version",
        ],
        env=environment,
        capture_output=True,
        text=True,
        check=False,
    )
    installed_version = version_check.stdout.strip()
    if version_check.returncode != 0:
        detail = version_check.stderr.strip() or "falha sem mensagem"
        raise SetupError(f"não foi possível validar o toolkit: {detail}")
    if installed_version != expected_version:
        raise SetupError(f"versão inesperada: {installed_version}")


def run() -> None:
    """Run the complete offline setup."""
    if not running_on_windows():
        raise SetupError("este instalador deve ser executado no Windows")
    if not supported_python():
        raise SetupError("é necessário Python 3.10 ou superior")

    root = Path(sys.argv[1] if len(sys.argv) > 1 else Path.cwd()).resolve()
    version = os.environ.get("MAINFRAME_TOOLKIT_VERSION", DEFAULT_VERSION)
    source = root / ".claude" / "skills" / "mainframe-jcl-migration"
    target = root / ".devin" / "skills" / "mainframe-jcl-migration-devin"
    mcp_config = root / ".devin" / "mcp_config.json"
    toolkit_source = find_toolkit_runtime(root, version)

    validate_runtime(toolkit_source, version)
    adapt_skill(source, target)
    write_mcp_config(mcp_config, toolkit_source)

    print("\nConfiguração concluída.")
    print("Skill: .devin/skills/mainframe-jcl-migration-devin/SKILL.md")
    print("MCP:   .devin/mcp_config.json")
    print(f"Versão validada: {version}")
    print("\nReinicie o Devin Desktop e invoque:")
    print("/mainframe-jcl-migration-devin <workspace> <JOB.jcl>")


if __name__ == "__main__":
    try:
        run()
    except SetupError as error:
        raise SystemExit(f"Erro: {error}") from None
