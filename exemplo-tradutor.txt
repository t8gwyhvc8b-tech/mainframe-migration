#!/usr/bin/env bash
set -euo pipefail

ROOT="${1:-$PWD}"
VERSION="${MAINFRAME_TOOLKIT_VERSION:-0.1.13}"

SOURCE="$ROOT/.claude/skills/mainframe-jcl-migration"
TARGET="$ROOT/.devin/skills/mainframe-jcl-migration-devin"
MCP_CONFIG="$ROOT/.devin/mcp_config.json"
TOOLKIT_SOURCE="$ROOT/src"

fail() {
  printf 'Erro: %s\n' "$1" >&2
  exit 1
}

[[ -f "$SOURCE/SKILL.md" ]] ||
  fail "skill MCP não encontrada em $SOURCE/SKILL.md"

case "$(uname -s)" in
  MINGW*|MSYS*|CYGWIN*) ;;
  *) fail "este instalador deve ser executado no Git Bash do Windows" ;;
esac

command -v cygpath >/dev/null ||
  fail "cygpath não encontrado; execute o script pelo Git Bash"

[[ -f "$TOOLKIT_SOURCE/mainframe_modernization_toolkit/__main__.py" ]] ||
  fail "código-fonte local do toolkit não encontrado em $TOOLKIT_SOURCE"

PYTHON=()
for candidate in python3.exe python.exe; do
  if command -v "$candidate" >/dev/null &&
      "$candidate" -c 'import sys; raise SystemExit(0 if sys.version_info >= (3, 10) else 1)' 2>/dev/null; then
    PYTHON=("$candidate")
    break
  fi
done

if ((${#PYTHON[@]} == 0)) && command -v py.exe >/dev/null &&
    py.exe -3 -c 'import sys; raise SystemExit(0 if sys.version_info >= (3, 10) else 1)' 2>/dev/null; then
  PYTHON=(py.exe -3)
fi

((${#PYTHON[@]} > 0)) ||
  fail "Python 3.10 ou superior não encontrado no PATH do Git Bash"

ROOT_WINDOWS="$(cygpath -aw "$ROOT")"
TOOLKIT_SOURCE_WINDOWS="$(cygpath -aw "$TOOLKIT_SOURCE")"

PYTHONPATH="$TOOLKIT_SOURCE_WINDOWS" "${PYTHON[@]}" -c \
  'from mainframe_modernization_toolkit.mcp_server import serve_mcp' 2>/dev/null ||
  fail "o Python disponível não contém a dependência mcp compatível; sem download, ela precisa estar previamente instalada"

"${PYTHON[@]}" - "$ROOT_WINDOWS" "$VERSION" <<'PY'
import json
from pathlib import Path
import re
import shutil
import sys

root = Path(sys.argv[1]).resolve()
version = sys.argv[2]
source = root / ".claude" / "skills" / "mainframe-jcl-migration"
target = root / ".devin" / "skills" / "mainframe-jcl-migration-devin"
mcp_path = root / ".devin" / "mcp_config.json"
toolkit_source = root / "src"

required_tools = {
    "mainframe_preflight",
    "mainframe_dependency_graph",
    "mainframe_get_callers",
    "mainframe_resolve_copybook",
    "mainframe_impact_analysis",
    "mainframe_run_migration_tool",
    "mainframe_query_artifact",
}

target.mkdir(parents=True, exist_ok=True)
shutil.copytree(source, target, dirs_exist_ok=True)
skill_path = target / "SKILL.md"
content = skill_path.read_text(encoding="utf-8")

missing = sorted(tool for tool in required_tools if tool not in content)
if missing:
    raise SystemExit(
        "A skill MCP de origem não contém as tools esperadas: "
        + ", ".join(missing)
    )

content, substitutions = re.subn(
    r"(?m)^name:\s*mainframe-jcl-migration\s*$",
    "name: mainframe-jcl-migration-devin",
    content,
    count=1,
)
if substitutions != 1:
    raise SystemExit("Não foi possível adaptar o nome da skill.")

if not re.search(r"(?m)^triggers:", content):
    content, substitutions = re.subn(
        r"(?m)^(argument-hint:[^\n]*\n)",
        r"\1triggers:\n  - user\n  - model\n",
        content,
        count=1,
    )
    if substitutions != 1:
        raise SystemExit("Não foi possível adicionar os triggers.")

skill_path.write_text(content, encoding="utf-8")
mcp_path.parent.mkdir(parents=True, exist_ok=True)

if mcp_path.exists():
    try:
        config = json.loads(mcp_path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as error:
        raise SystemExit(f"JSON inválido em {mcp_path}: {error}")
else:
    config = {}

if not isinstance(config, dict):
    raise SystemExit(f"{mcp_path} precisa conter um objeto JSON.")
servers = config.setdefault("mcpServers", {})
if not isinstance(servers, dict):
    raise SystemExit(f"'mcpServers' em {mcp_path} precisa ser um objeto.")

servers["mainframe-toolkit"] = {
    "command": sys.executable,
    "args": [
        "-m",
        "mainframe_modernization_toolkit",
        "mcp",
        "serve",
    ],
    "env": {
        "PYTHONPATH": str(toolkit_source),
    },
}

temporary = mcp_path.with_name(mcp_path.name + ".tmp")
temporary.write_text(
    json.dumps(config, indent=2, ensure_ascii=True) + "\n",
    encoding="utf-8",
)
temporary.replace(mcp_path)
json.loads(mcp_path.read_text(encoding="utf-8"))
PY

printf 'Validando pacote %s...\n' "$VERSION"

INSTALLED_VERSION="$(
  PYTHONPATH="$TOOLKIT_SOURCE_WINDOWS" "${PYTHON[@]}" \
        -m mainframe_modernization_toolkit --version |
    tr -d '\r'
)"

[[ "$INSTALLED_VERSION" == "$VERSION" ]] ||
  fail "versão inesperada: $INSTALLED_VERSION"

printf '\nConfiguração concluída.\n'
printf 'Skill: %s\n' \
  ".devin/skills/mainframe-jcl-migration-devin/SKILL.md"
printf 'MCP:   %s\n' ".devin/mcp_config.json"
printf 'Versão validada: %s\n' "$INSTALLED_VERSION"
printf '\nReinicie o Devin Desktop e invoque:\n'
printf '/mainframe-jcl-migration-devin <workspace> <JOB.jcl>\n'
case "$(uname -m)" in
  x86_64|amd64) UV_TARGET="x86_64-pc-windows-msvc" ;;
  arm64|aarch64) UV_TARGET="aarch64-pc-windows-msvc" ;;
  *) fail "arquitetura do Windows não suportada: $(uname -m)" ;;
esac

mkdir -p "$TARGET" "$BIN_DIR" "$UV_CACHE_DIR"
cp -R "$SOURCE/." "$TARGET/"

REQUIRED_TOOLS=(
    mainframe_preflight
    mainframe_dependency_graph
    mainframe_get_callers
    mainframe_resolve_copybook
    mainframe_impact_analysis
    mainframe_run_migration_tool
    mainframe_query_artifact
)

MISSING_TOOLS=()
for tool in "${REQUIRED_TOOLS[@]}"; do
    grep -Fq "$tool" "$TARGET/SKILL.md" || MISSING_TOOLS+=("$tool")
done

if ((${#MISSING_TOOLS[@]})); then
    fail "a skill MCP de origem não contém as tools esperadas: ${MISSING_TOOLS[*]}"
fi

[[ "$(grep -Ec '^name:[[:space:]]*mainframe-jcl-migration[[:space:]]*$' "$TARGET/SKILL.md")" == "1" ]] ||
    fail "não foi possível adaptar o nome da skill"

SKILL_TEMP="$TARGET/SKILL.md.tmp"
trap 'rm -f "$SKILL_TEMP" "$BIN_DIR/uv.zip.tmp" "$MCP_CONFIG.tmp"' EXIT

sed 's/^name:[[:space:]]*mainframe-jcl-migration[[:space:]]*$/name: mainframe-jcl-migration-devin/' \
    "$TARGET/SKILL.md" >"$SKILL_TEMP"
mv "$SKILL_TEMP" "$TARGET/SKILL.md"

if ! grep -Eq '^triggers:[[:space:]]*$' "$TARGET/SKILL.md"; then
    awk '
        { print }
        !inserted && /^argument-hint:/ {
            print "triggers:"
            print "  - user"
            print "  - model"
            inserted = 1
        }
        END { if (!inserted) exit 42 }
    ' "$TARGET/SKILL.md" >"$SKILL_TEMP" ||
        fail "não foi possível adicionar os triggers"
    mv "$SKILL_TEMP" "$TARGET/SKILL.md"
fi

if [[ ! -f "$UVX" ]]; then
    UV_ARCHIVE="$BIN_DIR/uv.zip.tmp"
    UV_URL="https://github.com/astral-sh/uv/releases/download/$UV_VERSION/uv-$UV_TARGET.zip"
    printf 'Baixando runtime portátil uv %s...\n' "$UV_VERSION"
    curl --fail --location --retry 3 --silent --show-error \
        --output "$UV_ARCHIVE" \
        "$UV_URL"
    tar -xf "$UV_ARCHIVE" -C "$BIN_DIR" ||
        fail "não foi possível extrair o runtime portátil com o tar do Git Bash"
    rm -f "$UV_ARCHIVE"
fi

[[ -f "$UVX" ]] ||
    fail "o pacote portátil do uv não forneceu $UVX"

json_escape() {
    sed 's/\\/\\\\/g; s/"/\\"/g'
}

UVX_WINDOWS="$(cygpath -aw "$UVX" | json_escape)"
UV_CACHE_WINDOWS="$(cygpath -aw "$UV_CACHE_DIR" | json_escape)"

cat >"$MCP_CONFIG.tmp" <<JSON
{
    "mcpServers": {
        "mainframe-toolkit": {
            "command": "$UVX_WINDOWS",
            "args": [
                "--from",
                "mainframe-modernization-toolkit==$VERSION",
                "mainframe-toolkit",
                "mcp",
                "serve"
            ],
            "env": {
                "UV_CACHE_DIR": "$UV_CACHE_WINDOWS"
            }
        }
    }
}
JSON
mv "$MCP_CONFIG.tmp" "$MCP_CONFIG"
trap - EXIT

printf 'Validando pacote %s...\n' "$VERSION"

INSTALLED_VERSION="$(
  UV_CACHE_DIR="$UV_CACHE_DIR" "$UVX" \
    --from "mainframe-modernization-toolkit==$VERSION" \
    mainframe-toolkit --version |
    tr -d '\r'
)"

[[ "$INSTALLED_VERSION" == "$VERSION" ]] ||
  fail "versão inesperada: $INSTALLED_VERSION"

printf '\nConfiguração concluída.\n'
printf 'Skill: %s\n' \
  ".devin/skills/mainframe-jcl-migration-devin/SKILL.md"
printf 'MCP:   %s\n' ".devin/mcp_config.json"
printf 'Versão validada: %s\n' "$INSTALLED_VERSION"
printf '\nReinicie o Devin Desktop e invoque:\n'
printf '/mainframe-jcl-migration-devin <workspace> <JOB.jcl>\n'
